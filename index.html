using UnityEngine; using System.Collections.Generic; // ========================================================================= // STICKMAN CLASH MASTER ARCHITECTURE // Central orchestrator that handles round systems, player spawning, and UI loops. // Attach this script to an empty GameObject in your scene named "GameManager". // ========================================================================= public class StickmanClashClone : MonoBehaviour { [Header("Game Loop Setup")] public GameObject player1Prefab; public GameObject player2Prefab; public Transform p1SpawnPoint; public Transform p2SpawnPoint; [Header("Score Configuration")] [SerializeField] private int roundsToWinMatch = 3; private int p1Score = 0; private int p2Score = 0; private GameObject spawnedP1; private GameObject spawnedP2; private bool roundEnding = false; private string centerStatusText = "GET READY!"; private static StickmanClashClone _instance; public static StickmanClashClone Instance { get { return _instance; } } void Awake() { if (_instance == null) _instance = this; else Destroy(gameObject); } void Start() { StartNewMatch(); } public void StartNewMatch() { p1Score = 0; p2Score = 0; StartCoroutine(ResetRoundRoutine("MATCH STARTING")); } public void PlayerDied(int playerNumber) { if (roundEnding) return; roundEnding = true; if (playerNumber == 1) { p2Score++; centerStatusText = "PLAYER 2 WINS ROUND!"; } else { p1Score++; centerStatusText = "PLAYER 1 WINS ROUND!"; } if (p1Score >= roundsToWinMatch) { centerStatusText = "PLAYER 1 IS THE CHAMPION!"; StartCoroutine(MatchEndRoutine()); } else if (p2Score >= roundsToWinMatch) { centerStatusText = "PLAYER 2 IS THE CHAMPION!"; StartCoroutine(MatchEndRoutine()); } else { StartCoroutine(ResetRoundRoutine(centerStatusText)); } } private System.Collections.IEnumerator ResetRoundRoutine(string announcement) { centerStatusText = announcement; if (spawnedP1 != null) Destroy(spawnedP1); if (spawnedP2 != null) Destroy(spawnedP2); WeaponPickup[] oldWeapons = FindObjectsByType(FindObjectsSortMode.None); foreach (WeaponPickup w in oldWeapons) Destroy(w.gameObject); yield return new WaitForSeconds(1.5f); if (player1Prefab != null && p1SpawnPoint != null) spawnedP1 = Instantiate(player1Prefab, p1SpawnPoint.position, Quaternion.identity); if (player2Prefab != null && p2SpawnPoint != null) spawnedP2 = Instantiate(player2Prefab, p2SpawnPoint.position, Quaternion.identity); if (spawnedP1 != null) { PlayerHealth h1 = spawnedP1.GetComponentInChildren(); if (h1 != null) h1.playerNumber = 1; } if (spawnedP2 != null) { PlayerHealth h2 = spawnedP2.GetComponentInChildren(); if (h2 != null) h2.playerNumber = 2; } centerStatusText = "FIGHT!"; yield return new WaitForSeconds(0.8f); centerStatusText = ""; roundEnding = false; } private System.Collections.IEnumerator MatchEndRoutine() { yield return new WaitForSeconds(3.0f); StartNewMatch(); } void OnGUI() { GUIStyle scoringStyle = new GUIStyle(); scoringStyle.fontSize = 28; scoringStyle.normal.textColor = Color.cyan; scoringStyle.alignment = TextAnchor.UpperLeft; GUI.Label(new Rect(20, 20, 300, 50), "P1 SCORE: " + p1Score, scoringStyle); scoringStyle.alignment = TextAnchor.UpperRight; scoringStyle.normal.textColor = Color.red; GUI.Label(new Rect(Screen.width - 320, 20, 300, 50), "P2 SCORE: " + p2Score, scoringStyle); if (!string.IsNullOrEmpty(centerStatusText)) { GUIStyle statusStyle = new GUIStyle(); statusStyle.fontSize = 42; statusStyle.fontStyle = FontStyle.Bold; statusStyle.normal.textColor = Color.yellow; statusStyle.alignment = TextAnchor.MiddleCenter; GUI.Label(new Rect(Screen.width / 2 - 400, Screen.height / 2 - 50, 800, 100), centerStatusText, statusStyle); } } } // ========================================================================= // MODULE 1: PLAYER PHYSICS CONTROLLER // ========================================================================= [RequireComponent(typeof(Rigidbody2D))] public class PlayerController : MonoBehaviour { [Header("Movement Settings")] [SerializeField] private float moveSpeed = 9f; [SerializeField] private float jumpForce = 13f; [SerializeField] private float airControlFactor = 0.85f; [Header("Ground Check")] [SerializeField] private Transform groundCheckPoint; [SerializeField] private float groundCheckRadius = 0.2f; [SerializeField] private LayerMask groundLayer; [Header("Weapon Rig Anchor")] public Transform handAnchor; private Rigidbody2D rb; private float horizontalInput; private bool isGrounded; private bool jumpRequested; private float facingDirection = 1f; private PlayerHealth health; void Start() { rb = GetComponent(); health = GetComponent(); rb.freezeRotation = true; rb.collisionDetectionMode = CollisionDetectionMode2D.Continuous; if (groundCheckPoint == null) { groundCheckPoint = this.transform; } } void Update() { if (health != null && health.playerNumber == 2) { horizontalInput = 0; if (Input.GetKey(KeyCode.LeftArrow)) horizontalInput = -1; if (Input.GetKey(KeyCode.RightArrow)) horizontalInput = 1; if (Input.GetKeyDown(KeyCode.UpArrow) && isGrounded) jumpRequested = true; if (Input.GetKeyDown(KeyCode.RightControl) && handAnchor != null && handAnchor.childCount > 0) TriggerWeaponAttack(); } else { horizontalInput = Input.GetAxisRaw("Horizontal"); if (Input.GetButtonDown("Jump") && isGrounded) jumpRequested = true; if (Input.GetButtonDown("Fire1") && handAnchor != null && handAnchor.childCount > 0) TriggerWeaponAttack(); } if (horizontalInput != 0) { facingDirection = Mathf.Sign(horizontalInput); transform.localScale = new Vector3(facingDirection, 1f, 1f); } } void TriggerWeaponAttack() { WeaponPickup activeWeapon = handAnchor.GetComponentInChildren(); if (activeWeapon != null) activeWeapon.UseWeapon(facingDirection); } void FixedUpdate() { isGrounded = Physics2D.OverlapCircle(groundCheckPoint.position, groundCheckRadius, groundLayer); float targetSpeed = horizontalInput * moveSpeed; if (!isGrounded) targetSpeed *= airControlFactor; rb.velocity = new Vector2(targetSpeed, rb.velocity.y); if (jumpRequested) { rb.velocity = new Vector2(rb.velocity.x, jumpForce); jumpRequested = false; } } } // ========================================================================= // MODULE 2A: PLAYER HEALTH SYSTEM // ========================================================================= public class PlayerHealth : MonoBehaviour { public int playerNumber = 1; [SerializeField] private float maxHealth = 100f; private float currentHealth; public bool isDead { get; private set; } void Start() { currentHealth = maxHealth; isDead = false; } public void TakeDamage(float amount) { if (isDead) return; currentHealth -= amount; if (currentHealth <= 0) { isDead = true; if (StickmanClashClone.Instance != null) { StickmanClashClone.Instance.PlayerDied(playerNumber); } } } } public class WeaponPickup : MonoBehaviour { public void UseWeapon(float direction) { } }
